    35→/* Inline citations (Step 16.4): turn LLM-emitted [n] markers and any source
    36→   filename mentioned in the answer into clickable links to that document.
    37→   Deterministic — works even when the LLM never emits a marker. */
    38→function renderAnswerContent(
    39→  content: string,
    40→  sources: ChatSource[] | undefined,
    41→  onOpenDoc: (s: ChatSource) => void
    42→): ReactNode {
    43→  if (!sources || sources.length === 0) return content;
    44→
    45→  const filenames = sources
    46→    .map((s) => s.filename)
    47→    .filter(Boolean)
    48→    .sort((a, b) => b.length - a.length)
    49→    .map(escapeRegExp);
    50→  const nameGroup = filenames.length > 0 ? `|(${filenames.join("|")})` : "";
    51→  const pattern = new RegExp(`\\[(\\d{1,2})\\]${nameGroup}`, "g");
    52→
    53→  const nodes: ReactNode[] = [];
    54→  let last = 0;
    55→  let key = 0;
    56→  for (const m of Array.from(content.matchAll(pattern))) {
    57→    const idx = m.index ?? 0;
    58→    if (idx > last) nodes.push(content.slice(last, idx));
    59→    if (m[1] !== undefined) {
    60→      const n = parseInt(m[1], 10);
    61→      const s = sources[n - 1];
    62→      if (s) {
    63→        nodes.push(
    64→          <button
    65→            key={key++}
    66→            onClick={() => onOpenDoc(s)}
    67→            title={s.filename}
    68→            className="inline-flex items-center justify-center rounded bg-primary/15 text-primary font-semibold px-1 mx-0.5 text-[11px] leading-4 hover:bg-primary/25 transition-colors"
    69→          >
    70→            [{n}]
    71→          </button>
    72→        );
    73→      } else {
    74→        nodes.push(m[0]); // out-of-range marker stays plain text
    75→      }
    76→    } else {
    77→      const s = sources.find((x) => x.filename === m[0]);
    78→      if (s) {
    79→        nodes.push(
    80→          <button
    81→            key={key++}
    82→            onClick={() => onOpenDoc(s)}
    83→            className="text-primary font-medium underline decoration-primary/40 underline-offset-2 hover:decoration-primary transition-all break-all"
    84→          >
    85→            {m[0]}
    86→          </button>
    87→        );
    88→      } else {
    89→        nodes.push(m[0]);
    90→      }
    91→    }
    92→    last = idx + m[0].length;
    93→  }
    94→  if (last < content.length) nodes.push(content.slice(last));
    95→  return nodes;
    96→}

    98→/* Authenticated thumbnail for a visual citation (same loader contract as the
    99→   search page): fetch with bearer token, render as an object URL, clean up. */
   100→function ChatVisualThumb({ assetId, alt }: { assetId: number; alt: string }) {
   101→  const [url, setUrl] = useState<string | null>(null);
   102→  const [failed, setFailed] = useState(false);
   103→
   104→  useEffect(() => {
   105→    let active = true;
   106→    let objectUrl: string | null = null;
   107→    getVisualThumbnailUrl(assetId)
   108→      .then((u) => {
   109→        if (active) {
   110→          objectUrl = u;
   111→          setUrl(u);
   112→        } else {
   113→          URL.revokeObjectURL(u);
   114→        }
   115→      })
   116→      .catch(() => {
   117→        if (active) setFailed(true);
   118→      });
   119→    return () => {
   120→      active = false;
   121→      if (objectUrl) URL.revokeObjectURL(objectUrl);
   122→    };
   123→  }, [assetId]);
   124→
   125→  if (failed) {
   126→    return (
   127→      <div className="w-14 h-14 rounded-lg bg-surface border border-border flex items-center justify-center">
   128→        <ImageIcon className="w-5 h-5 text-muted-foreground" />
   129→      </div>
   130→    );
   131→  }
   132→  if (!url) {
   133→    return <div className="w-14 h-14 rounded-lg bg-surface border border-border animate-pulse" />;
   134→  }
   135→  return <img src={url} alt={alt} className="w-14 h-14 rounded-lg object-cover border border-border" />;
   136→}
