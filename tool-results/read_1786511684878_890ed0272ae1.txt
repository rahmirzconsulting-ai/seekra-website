   465→export function logout(): void {
   466→  if (typeof window === 'undefined') return;
   467→  localStorage.removeItem(TOKEN_KEY);
   468→  localStorage.removeItem(LEGACY_TOKEN_KEY);
   469→  // hygiene: no user data may survive logout (shared-machine safety)
   470→  sessionStorage.removeItem('seekra_search_state');

   472→  fileUrlCache.forEach((url) => URL.revokeObjectURL(url));
   473→  fileUrlCache.clear();
   474→}

   476→/** FastAPI errors: detail may be a string OR an array of {msg} objects
   477→ *  (422 validation). Stringifying the array produced "[object Object]". */
   478→function normalizeDetail(errorData: any, fallback: string): string {
   479→  const d = errorData?.detail;
   480→  if (!d) return fallback;
   481→  if (typeof d === 'string') return d;
   482→  if (Array.isArray(d)) {
   483→    return d
   484→      .map((e: any) => e?.msg || JSON.stringify(e))
   485→      .join('; ');
   486→  }
   487→  return String(d);
   488→}

   490→async function apiFetch<T>(
   491→  endpoint: string,
   492→  options: RequestInit = {}
   493→): Promise<T> {
   494→  const token = getToken();
   495→  const headers: Record<string, string> = {
   496→    ...(options.headers as Record<string, string>),
   497→  };
   498→
   499→  if (token) headers['Authorization'] = `Bearer ${token}`;
   500→  if (options.body && !(options.body instanceof FormData)) {
   501→    headers['Content-Type'] = 'application/json';
   502→  }
   503→
   504→  const response = await fetch(`${API_BASE}${endpoint}`, {
   505→    ...options,
   506→    headers,
   507→  });
   508→
   509→  if (!response.ok) {
   510→    const errorData = await response.json().catch(() => ({}));
   511→    throw new Error(normalizeDetail(errorData, `HTTP ${response.status}`));
   512→  }
   513→
   514→  return response.json();
   515→}

   519→export async function login(credentials: LoginCredentials): Promise<AuthResponse> {
   520→  const formData = new URLSearchParams();
   521→  formData.append('username', credentials.username);
   522→  formData.append('password', credentials.password);
   523→
   524→  const response = await fetch(`${API_BASE}/auth/login`, {
   525→    method: 'POST',
   526→    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   527→    body: formData.toString(),
   528→  });
   529→
   530→  if (!response.ok) {
   531→    const errorData = await response.json().catch(() => ({}));
   532→    throw new Error(normalizeDetail(errorData, 'Login failed'));
   533→  }
   534→
   535→  const data = await response.json();
   536→  if (typeof window !== 'undefined') {
   537→    localStorage.removeItem(LEGACY_TOKEN_KEY);
   538→    localStorage.setItem(TOKEN_KEY, data.access_token);
   539→  }
   540→  return data;
   541→}

   635→export async function downloadFile(docId: number, _filename?: string, _preview?: boolean, rendition?: string): Promise<Blob> {
   636→  const token = getToken();
   637→  const qs = rendition ? `?rendition=${encodeURIComponent(rendition)}` : '';
   638→  const response = await fetch(`${API_BASE}/files/download/${docId}${qs}`, {
   639→    headers: token ? { Authorization: `Bearer ${token}` } : {},
   640→  });
   641→  if (!response.ok) throw new Error(`Download failed: ${response.status}`);
   642→  return response.blob();
   643→}

   645→/** Authenticated object URL for embedding PDFs/media in the viewer.
   646→ *  Cached in-memory: repeat opens are instant. */
   647→const fileUrlCache = new Map<string, string>();
   648→export async function getFileUrl(docId: number, rendition?: string): Promise<string> {
   649→  const key = rendition ? `${docId}:r${rendition}` : String(docId);
   650→  const hit = fileUrlCache.get(key);
   651→  if (hit) return hit;
   652→  const blob = await downloadFile(docId, undefined, undefined, rendition);
   653→  const url = URL.createObjectURL(blob);
   654→  fileUrlCache.set(key, url);
   655→  return url;
   656→}

   658→/** Warm the cache in the background (call with top search results). */
   659→export function prefetchFileUrls(docIds: number[]): void {
   660→  docIds.forEach((id) => {
   661→    if (!fileUrlCache.has(String(id))) {
   662→      getFileUrl(id).catch(() => {});
   663→    }
   664→  });
   665→}

   677→export async function search(data: SearchRequest): Promise<SearchResponse> {
   678→  const result = await apiFetch<any>('/search/', {
   679→    method: 'POST',
   680→    body: JSON.stringify({
   681→      query: data.query,
   682→      limit: data.limit || 10,
   683→      ...(data.collection_id != null ? { collection_id: data.collection_id } : {}),
   684→      ...(data.document_ids?.length ? { document_ids: data.document_ids } : {}),
   685→    }),
   686→  });
   687→  const results: SearchResult[] = (result.results || []).map((r: any) => ({
   688→    ...r,
   689→    id: r.document_id,
   690→    mime_type: r.content_type || null,
   691→  }));
   692→  return {
   693→    query: result.query,
   694→    results,
   695→    total: result.total ?? results.length,
   696→    count: result.total ?? results.length,
   697→    legs: result.legs ?? { vector_hits: 0, keyword_hits: 0, segment_hits: 0 },
   698→  };
   699→}

   763→/* Step 30: neural spoken answers (edge-tts). Returns an object URL for the
   764→   MP3 stream; throws on failure so the caller can fall back to browser
   765→   speech. The caller owns URL.revokeObjectURL. */
   766→export async function speakText(text: string, language?: string): Promise<string> {
   767→  const token = getToken();
   768→  const headers: Record<string, string> = { 'Content-Type': 'application/json' };
   769→  if (token) headers['Authorization'] = `Bearer ${token}`;
   770→  const response = await fetch(`${API_BASE}/chat/tts`, {
   771→    method: 'POST',
   772→    headers,
   773→    body: JSON.stringify({ text, language }),
   774→  });
   775→  if (!response.ok) throw new Error(`HTTP ${response.status}`);
   776→  return URL.createObjectURL(await response.blob());
   777→}
