   131→def detect_language(text: str) -> str:
   132→    sample = (text or "")[:1000]
   133→    # Step 30E: Urdu-specific Nastaliq letters (never used in Arabic):
   134→    # ٹ ڈ ڑ ں ھ ہ ے
   135→    urdu = len(re.findall(r"[\u0679\u0688\u0691\u06BA\u06BE\u06C1\u06D2]", sample))
   136→    if urdu > 0:
   137→        return "ur"
   138→    arabic = len(re.findall(r"[\u0600-\u06FF]", sample))
   139→    latin = len(re.findall(r"[a-zA-Z]", sample))
   140→    return "ar" if arabic > latin else "en"

   155→def _looks_like_followup(question: str) -> bool:
   156→    words = question.split()
   157→    if len(words) <= 6:

   163→def _format_history(history: list[dict], max_chars: int = 1200) -> str:
   164→    lines = []
   165→    for turn in history[-HISTORY_TURNS:]:
   166→        role = "User" if (turn.get("role") or "") == "user" else "Assistant"
   167→        content = " ".join(str(turn.get("content") or "").split())[:400]
   168→        if content:
   169→            lines.append(f"{role}: {content}")
   170→    text = "\n".join(lines)
   171→    return text[-max_chars:] if len(text) > max_chars else text

   174→async def _contextualize_question(question: str, history: list[dict], lang: str) -> str:
   175→    """Rewrite a follow-up into a standalone retrieval query (fail-open)."""
   176→    if not history or not _looks_like_followup(question):
   177→        return question
   178→    lang_name = LANG_NAMES.get(lang, "English")
   179→    prompt = (
   180→        "You rewrite follow-up questions into standalone search questions for a "
   181→        "document search system. Given the conversation and the follow-up, output "
   182→        "ONLY the rewritten standalone question in " + lang_name + ". Resolve "
   183→        "references like 'this book', 'it', 'that document' to the specific "
   184→        "document or topic named earlier in the conversation. Do NOT answer the "
   185→        "question. If the follow-up is already standalone, output it unchanged.\n\n"
   186→        "Conversation:\n" + _format_history(history) + "\n\n"
   187→        "Follow-up: " + question + "\n\nStandalone question:"
   188→    )
   189→    try:
   190→        rewritten = await asyncio.wait_for(
   191→            llm_gateway._generate(prompt, max_tokens=60, kind="chat"),
   192→            timeout=REWRITE_TIMEOUT)
   193→        rewritten = unicodedata.normalize("NFKC", (rewritten or "").strip().strip('"').strip())
   194→        first_line = rewritten.splitlines()[0].strip() if rewritten else ""
   195→        if first_line and len(first_line) <= 400:
   196→            return first_line
   197→    except Exception:
   198→        pass  # timeout / provider error -> original question
   199→    return question

   211→def _type_label(content_type: str, filename: str = "") -> str:
   212→    ct = (content_type or "").lower()
   213→    ext = (filename.rsplit(".", 1)[-1] if "." in filename else "").lower()
   214→    if "pdf" in ct or ext == "pdf":
   215→        return "PDF"
   216→    if ct.startswith("image") or ct == "images" or ext in ("jpg", "jpeg", "png", "gif", "webp", "bmp"):
   217→        return "images"
   218→    if ct.startswith("audio") or ct == "audio" or ext in ("mp3", "wav", "m4a", "ogg", "flac"):
   219→        return "audio"
   220→    if ct.startswith("video") or ct == "video" or ext in ("mp4", "mov", "webm", "mkv", "avi"):
   221→        return "video"
   222→    if any(x in ct for x in ("word", "presentation", "sheet", "officedocument")) or ct == "office" \
   223→            or ext in ("doc", "docx", "ppt", "pptx", "xls", "xlsx", "csv"):
   224→        return "Office"
   225→    if ct.startswith("text") or ext in ("txt", "md"):
   226→        return "text"
   227→    return "other"

   230→async def resolve_target_documents(question: str, db: AsyncSession, user=None) -> list[int]:
   231→    """Resolve meta-references: 'latest document' intent + filename mentions."""
   232→    q_lower = question.lower()
   233→    perm_params: dict = {}
   234→    perm_sql = permission_clause(user, perm_params, "documents")
   235→
   236→    # 1. "latest / newest document" intent (EN + AR)
   237→    if any(m in q_lower for m in LATEST_MARKERS):
   238→        result = await db.execute(sql_text(
   239→            f"SELECT id FROM documents WHERE {perm_sql} ORDER BY created_at DESC LIMIT 1"), perm_params)
   240→        row = result.first()
   241→        return [row[0]] if row else []
   242→
   243→    # 2. Filename mention detection (distinctive tokens, e.g. "honda", "mant aq")
   244→    result = await db.execute(sql_text(f"SELECT id, filename FROM documents WHERE {perm_sql}"), perm_params)
   245→    q_tokens = set(re.findall(r"\w+", q_lower)) - STOPWORDS
   246→    matched = []
   247→    for row in result.fetchall():
   248→        name_tokens = set(re.sub(r"[._\-]", " ", row.filename.lower()).split()) - STOPWORDS
   249→        overlap = name_tokens & q_tokens
   250→        distinctive = {t for t in overlap if len(t) >= 4 and t not in GENERIC_NAME_WORDS}
   251→        # scope only on strong evidence: >=2 shared tokens, or one distinctive token
   252→        if len(overlap) >= 2 or (len(overlap) == 1 and distinctive):
   253→            matched.append(row.id)
   254→    return matched

   258→    """Distinct detected object labels (shared cache key with smart search)."""

   262→        return {str(label).lower() for label in cached}

   264→        result = await session.execute(
   265→            sql_text("SELECT DISTINCT lower(label) AS label FROM visual_labels"))

   267→    await cache_set_json(cache_key, labels, LABEL_CACHE_TTL)

   283→async def warm_visual_model() -> None:
   284→    """Preload the visual embedding model at startup (Step 16.7).
   285→
   286→    The first embed call after a backend boot loads the model weights
   287→    (multi-second on CPU). Without warmup that cost lands inside the
   288→    per-answer timeout, so the first visual question gets exact-leg-only
   289→    results while later questions get full semantic results — the same
   290→    question appears to return different sources. Warming at startup makes
   291→    every question run the identical full pipeline."""
   292→    try:
   293→        async with _VISUAL_MODEL_LOCK:
   294→            await asyncio.to_thread(embed_visual_text, "seekra visual model warmup")
   295→        print("[CHAT] visual model warmup complete")
   296→    except Exception as exc:
   297→        print(f"[CHAT] visual model warmup failed: {type(exc).__name__}")

   307→async def _visual_ocr_leg(question: str, target_ids: list[int], fetch: int = 8, user=None) -> list[dict]:
   308→    params: dict = {"query": question, "fetch": fetch}
   309→    scope = _visual_scope(target_ids, params)
   310→    scope += " AND (" + permission_clause(user, params, "d") + ")"
   311→    async with async_session() as session:
   312→        rows = (await session.execute(sql_text(f"""
   313→            SELECT va.id AS asset_id, va.document_id, va.asset_type, va.page_number,
   314→                   va.start_time, va.end_time, va.ocr_text,
   315→                   ts_rank(to_tsvector('simple', coalesce(va.ocr_text, '')),
   316→                           websearch_to_tsquery('simple', :query)) AS score,
   317→                   d.filename
   318→            FROM visual_assets va
   319→            JOIN documents d ON d.id = va.document_id
   320→            WHERE COALESCE(d.status, 'completed') = 'completed'
   321→              AND to_tsvector('simple', coalesce(va.ocr_text, '')) @@ websearch_to_tsquery('simple', :query){scope}
   322→            ORDER BY score DESC, va.id
   323→            LIMIT :fetch
   324→        """), params)).mappings().all()
   325→    return [dict(r) for r in rows]

   328→async def _visual_object_leg(object_terms: list[str], target_ids: list[int], fetch: int = 8, user=None) -> list[dict]:
   329→    params: dict = {"labels": object_terms, "fetch": fetch}
   330→    scope = _visual_scope(target_ids, params)
   331→    scope += " AND (" + permission_clause(user, params, "d") + ")"
   332→    async with async_session() as session:
   333→        rows = (await session.execute(sql_text(f"""
   334→            SELECT va.id AS asset_id, va.document_id, va.asset_type, va.page_number,
   335→                   va.start_time, va.end_time, va.ocr_text,
   336→                   MAX(vl.confidence) AS score,
   337→                   d.filename
   338→            FROM visual_assets va
   339→            JOIN visual_labels vl ON vl.asset_id = va.id
   340→            JOIN documents d ON d.id = va.document_id
   341→            WHERE COALESCE(d.status, 'completed') = 'completed'
   342→              AND lower(vl.label) = ANY(CAST(:labels AS text[])){scope}
   343→            GROUP BY va.id, d.id
   344→            ORDER BY score DESC, va.id
   345→            LIMIT :fetch
   346→        """), params)).mappings().all()
   347→    return [dict(r) for r in rows]

   350→async def _visual_vector_leg(question: str, target_ids: list[int], fetch: int = 8, user=None) -> list[dict]:
   351→    embedding = await _cached_visual_embedding(question)
   352→    if not embedding:
   353→        return []
   354→    params: dict = {"emb": str(embedding), "fetch": fetch}
   355→    scope = _visual_scope(target_ids, params)
   356→    scope += " AND (" + permission_clause(user, params, "d") + ")"
   357→    async with async_session() as session:
   358→        rows = (await session.execute(sql_text(f"""
   359→            SELECT va.id AS asset_id, va.document_id, va.asset_type, va.page_number,
   360→                   va.start_time, va.end_time, va.ocr_text,
   361→                   1 - (va.embedding <=> CAST(:emb AS vector)) AS score,
   362→                   d.filename
   363→            FROM visual_assets va
   364→            JOIN documents d ON d.id = va.document_id
   365→            WHERE COALESCE(d.status, 'completed') = 'completed'
   366→              AND va.embedding IS NOT NULL{scope}
   367→            ORDER BY va.embedding <=> CAST(:emb AS vector)
   368→            LIMIT :fetch
   369→        """), params)).mappings().all()
   370→    return [dict(r) for r in rows if float(r["score"] or 0) >= VISUAL_SIM_FLOOR]

   373→async def _visual_evidence_inner(question: str, target_ids: list[int], has_text_rows: bool, user=None) -> tuple[list[dict], dict[str, int]]:
   374→    """OCR + object + (gated) semantic visual legs, RRF-fused with smart-search
   375→    qualification rules. The semantic leg only runs on visual intent or when the
   376→    text legs found nothing — the visual model never slows down a text answer."""
   377→    try:
   378→        known_labels = await _known_visual_labels()
   379→    except Exception:
   380→        known_labels = set()
   381→    q_lower = question.lower()
   382→    q_terms = {t for t in re.findall(r"\w+", q_lower) if t not in STOPWORDS}
   383→    object_terms = sorted(q_terms & known_labels)
   384→    # Step 16.9: listing questions ("which files show X?") count as visual
   385→    # intent — otherwise the semantic leg stays off whenever text rows exist
   386→    # and the listing path never sees the visual matches (V3 regression).
   387→    visual_intent = bool(object_terms or _is_visual_listing(q_lower)
   388→                         or any(term in q_lower for term in VISUAL_INTENT_TERMS))
   389→
   390→    # Exact legs (OCR + detected objects): fast indexed SQL, always collected.
   391→    exact_legs = ["ocr"]
   392→    exact_tasks = [_visual_ocr_leg(question, target_ids, user=user)]
   393→    if object_terms:
   394→        exact_legs.append("object")
   395→        exact_tasks.append(_visual_object_leg(object_terms, target_ids, user=user))
   396→    exact_results = await asyncio.gather(*exact_tasks, return_exceptions=True)
   397→    leg_rows = {leg: rows for leg, rows in zip(exact_legs, exact_results) if isinstance(rows, list)}
   398→
   399→    # Semantic leg: isolated timeout — a cold visual model (first question after
   400→    # a backend restart) must never cancel the exact legs that already answered.
   401→    if visual_intent or not has_text_rows:
   402→        try:
   403→            leg_rows["vector"] = await asyncio.wait_for(
   404→                _visual_vector_leg(question, target_ids, user=user), timeout=VISUAL_TIMEOUT)
   405→        except Exception:
   406→            leg_rows["vector"] = []
   407→
   408→    leg_counts = {leg: len(rows) for leg, rows in leg_rows.items()}
   409→
   410→    assets: dict[int, dict] = {}
   411→    for leg_name in ("ocr", "object", "vector"):
   412→        for rank, row in enumerate(leg_rows.get(leg_name, []), start=1):
   413→            asset_id = int(row["asset_id"])
   414→            item = assets.setdefault(asset_id, {
   415→                "asset_id": asset_id, "document_id": int(row["document_id"]),
   416→                "filename": row["filename"], "asset_type": row.get("asset_type"),
   417→                "page_number": row.get("page_number"), "start_time": row.get("start_time"),
   418→                "ocr_text": row.get("ocr_text") or "", "score": 0.0,
   419→                "labels": [], "_legs": set(), "_vector_scores": []})
   420→            if leg_name not in item["_legs"]:
   421→                item["_legs"].add(leg_name)
   422→                item["score"] += _rrf(rank)
   423→            if leg_name == "vector":
   424→                item["_vector_scores"].append(float(row.get("score") or 0))
   425→
   426→    top_vector = max((s for item in assets.values() for s in item["_vector_scores"]), default=0.0)
   427→    semantic_floor = max(VISUAL_SEMANTIC_ONLY_FLOOR, top_vector - VISUAL_RELATIVE_DROP)
   428→    qualified = []
   429→    for item in assets.values():
   430→        exact = bool(item["_legs"] & {"ocr", "object"})
   431→        best_vector = max(item["_vector_scores"], default=0.0)
   432→        if not exact and best_vector < semantic_floor:
   433→            continue  # weak semantic-only visual hit (Step 14.4 parity)
   434→        item["exact"] = exact  # Step 16.8: certainty flag survives leg teardown
   435→        qualified.append(item)
   436→    # Stability (Step 16.7): exact hits (OCR text / detected object labels) are
   437→    # certain matches — they pin above the VISUAL_K cut and can never be
   438→    # displaced by semantic-only vector guesses, which only fill remaining
   439→    # slots. Deterministic tiebreaks keep repeated identical questions
   440→    # identical whether the vector leg ran warm, cold, or not at all.
   441→    qualified.sort(key=lambda item: (
   442→        0 if item["_legs"] & {"ocr", "object"} else 1,
   443→        -item["score"],
   444→        item["asset_id"],
   445→    ))
   446→    # Step 16.9: document diversity — each distinct document takes one slot
   447→    # first, then remaining slots fill by rank. A stack of near-duplicate
   448→    # frames from one video can no longer crowd out distinct matching
   449→    # documents (the meeting photo behind three meeting video frames).
   450→    _first_pass, _second_pass, _seen_docs = [], [], set()
   451→    for item in qualified:
   452→        if item["document_id"] not in _seen_docs:
   453→            _seen_docs.add(item["document_id"])
   454→            _first_pass.append(item)
   455→        else:
   456→            _second_pass.append(item)
   457→    qualified = (_first_pass + _second_pass)[:VISUAL_K]
   458→    if not qualified:
   459→        return [], leg_counts, visual_intent
   460→
   461→    async with async_session() as session:
   462→        label_rows = (await session.execute(sql_text("""
   463→            SELECT asset_id, label, confidence
   464→            FROM visual_labels
   465→            WHERE asset_id = ANY(CAST(:asset_ids AS int[]))
   466→            ORDER BY asset_id, confidence DESC
   467→        """), {"asset_ids": [item["asset_id"] for item in qualified]})).mappings().all()
   468→    labels: dict[int, list[str]] = {}
   469→    for row in label_rows:
   470→        entries = labels.setdefault(int(row.asset_id), [])
   471→        if len(entries) < 4 and row.label not in entries:
   472→            entries.append(row.label)
   473→    for item in qualified:
   474→        item["labels"] = labels.get(item["asset_id"], [])
   475→        item["score"] = round(item["score"], 6)
   476→        item.pop("_legs", None)
   477→        item.pop("_vector_scores", None)
   478→    return qualified, leg_counts, visual_intent

   481→async def _chat_visual_evidence(question: str, target_ids: list[int], has_text_rows: bool, user=None) -> list[dict]:
   482→    """Visual grounding for one chat answer; never raises, never blocks.
   483→    Always logs one line so visual behavior is diagnosable from backend logs."""
   484→    started = time.perf_counter()
   485→    try:
   486→        hits, leg_counts, visual_intent = await asyncio.wait_for(
   487→            _visual_evidence_inner(question, target_ids, has_text_rows, user),
   488→            timeout=VISUAL_TIMEOUT + 1.5,  # exact legs budget; vector leg has its own
   489→        )
   490→        ms = int((time.perf_counter() - started) * 1000)
   491→        print(f"[CHAT] visual_ms={ms} visual_legs={leg_counts} visual_hits={len(hits)}")
   492→        return hits, visual_intent
   493→    except Exception as exc:
   494→        print(f"[CHAT] visual grounding unavailable: {type(exc).__name__}")
   495→        return [], False  # missing tables / DB error -> text-only answer

   823→                source["pii"] = pii_for_document(pii_map, source["document_id"])
   824→            pii_notice = build_pii_notice(pii_map, [s["document_id"] for s in sources])
   825→            await audit_event(
   826→                db, None, current_user.id, "CHAT_VISUAL_LISTING",
   827→                f"{current_user.username} visual listing: {len(list_docs)} document(s) matched",
   828→            )
   829→            return {"answer": "\n".join(lines), "question": req.question,
   830→                    "sources": sources, "pii_notice": pii_notice}
   831→
   832→        # 4. Ground the answer in the fused top-K hits (page / timestamp labeled)
   833→        def _ts_label(sec):
   834→            sec = float(sec or 0)
   835→            return f"{int(sec // 60)}:{int(sec % 60):02d}"
   836→
   837→        # Citation relevance (Step 16.3): the LLM context keeps broad recall, but
   838→        # displayed citations need strong evidence — a document qualifies only if
   839→        # its best chunk similarity sits within SOURCE_REL_DROP of the top hit
   840→        # (Step 14.4-style relative band). No vector evidence -> top fused rows.
   841→        sims = [float(r["similarity"]) for r in rows if r.get("similarity") is not None]
   842→        if sims:
   843→            band = max(SIM_FLOOR, max(sims) - SOURCE_REL_DROP)
   844→            cited = [r for r in rows
   845→                     if r.get("similarity") is not None and float(r["similarity"]) >= band]
   846→        else:
   847→            cited = rows[:3]  # keyword-only mode (embedding unavailable)
   848→        if not cited:
   849→            cited = rows[:1]
   850→
   851→        # Citation numbering (Step 16.4): context documents are numbered exactly
   852→        # like the displayed sources so the LLM can cite inline as [1], [2] and the
   853→        # frontend can map each marker to its source (same dedupe rules, same order).
   854→        cite_order: dict[int, int] = {}
   855→        _seen_fn_num: set = set()
   856→        for r in cited:
   857→            if r["filename"] in _seen_fn_num:
   858→                continue
   859→            if r["document_id"] not in cite_order:
   860→                cite_order[r["document_id"]] = len(cite_order) + 1
   861→                _seen_fn_num.add(r["filename"])
   862→        for v in visual_hits:
   863→            if v["document_id"] not in cite_order and v["filename"] not in _seen_fn_num:
   864→                cite_order[v["document_id"]] = len(cite_order) + 1
   865→                _seen_fn_num.add(v["filename"])
   866→
   867→        def _ctx_line(r):
   868→            if r["page_number"]:
   869→                loc = f" p.{r['page_number']}"
   870→            elif r["start_time"] is not None:
   871→                loc = f" @ {_ts_label(r['start_time'])}"
   872→            else:
   873→                loc = ""
   874→            n = cite_order.get(r["document_id"])
   875→            tag = f"[Doc {n}: " if n else "[Document: "
   876→            return f"{tag}{r['filename']}{loc}]\n{r['chunk_text']}"
   877→
   878→        def _visual_ctx_line(v):
   879→            # Type-accurate tags (Step 16.6): the LLM must know whether a visual
   880→            # hit is an image, a video moment, or a PDF page — otherwise it calls
   881→            # video frames "pictures" in the answer.
   882→            atype = (v.get("asset_type") or "image").lower()
   883→            if atype == "video_frame":
   884→                kind = "video"
   885→                loc = f" @ {_ts_label(v['start_time'])}" if v.get("start_time") is not None else ""
   886→                text_label = "Text in frame"
   887→            elif atype == "pdf_page":
   888→                kind = "PDF page"
   889→                loc = f" p.{v['page_number']}" if v.get("page_number") else ""
   890→                text_label = "Text on page"
   891→            else:
   892→                kind = "image"
   893→                loc = ""
   894→                text_label = "Text in image"
   895→            labels = ", ".join(v.get("labels") or [])
   896→            n = cite_order.get(v["document_id"])
   897→            tag = f"[Doc {n} ({kind}): " if n else f"[{kind.capitalize()}: "
   898→            header = f"{tag}{v['filename']}{loc}" + (f" | objects: {labels}" if labels else "") + "]"
   899→            ocr = " ".join((v.get("ocr_text") or "").split())[:400]
   900→            return header + (f"\n{text_label}: {ocr}" if ocr else "")
   901→
   902→        context = "\n\n---\n\n".join(
   903→            [_ctx_line(r) for r in rows] + [_visual_ctx_line(v) for v in visual_hits]
   904→        )
   905→        if cite_order:
   906→            if lang == "ar":
   907→                question_for_llm = question + (
   908→                    "\n\nعند استخدامك معلومات من مستند مرقّم، اذكر المصدر داخل الإجابة "
   909→                    "بالشكل [1] أو [2] مطابقًا لرقم المستند — أرقامًا فقط، "
   910→                    "مثال: «...كما ورد في السياسة [1].»"
   911→                )
   912→            else:
   913→                question_for_llm = question + (
   914→                    "\n\nWhen you use information from a numbered document, cite it inline "
   915→                    "as [1], [2], etc. matching the document number — numbers only, "
   916→                    "e.g. '...as stated in the policy [1].'"
   917→                )
   918→        else:
   919→            question_for_llm = question
   920→        if history:
   921→            hist_label = "\u0627\u0644\u0645\u062d\u0627\u062f\u062b\u0629 \u062d\u062a\u0649 \u0627\u0644\u0622\u0646" if lang == "ar" else "Conversation so far"
   922→            question_for_llm = (
   923→                f"{hist_label}:\n" + _format_history(history) +
   924→                "\n\n" + question_for_llm
   925→            )
   926→        answer = await llm_gateway.answer_question(question_for_llm, context, lang)
   927→
   928→        # Step 16.8: despite the "numbers only" instruction the LLM sometimes
   929→        # writes "[Doc 1: file.pdf]" — normalize to bare [1] markers so the
   930→        # answer-verified filter and the frontend linkifier both work.
   931→        answer = re.sub(r"\[\s*Doc(?:ument)?\s+(\d{1,2})[^\]]*\]", r"[\1]", answer)
   932→
   933→
   934→        # 5. Citations: pages of the chunks that grounded the answer (relevance order)
   935→        seen, seen_fn, sources = set(), set(), []
   936→        for r in cited:
   937→            if r["filename"] in seen_fn:
   938→                continue
   939→            if r["document_id"] not in seen:
   940→                seen.add(r["document_id"])
   941→                seen_fn.add(r["filename"])
   942→                sc = r["similarity"] if r["similarity"] is not None else round(r["rrf"], 3)
   943→                sources.append({"document_id": r["document_id"], "filename": r["filename"],
   944→                                "score": round(float(sc), 3),
   945→                                "pages": [], "start_time": None})
   946→        for r in cited:
   947→            src_doc = next((s for s in sources if s["document_id"] == r["document_id"]), None)
   948→            if src_doc is None:
   949→                continue  # duplicate-upload doc collapsed by filename dedupe
   950→            if r["page_number"] is not None:
   951→                if r["page_number"] not in src_doc["pages"] and len(src_doc["pages"]) < 3:
   952→                    src_doc["pages"].append(int(r["page_number"]))
   953→            if r["start_time"] is not None and src_doc["start_time"] is None:
   954→                src_doc["start_time"] = r["start_time"]
   955→
   956→        # A/V citations: best-matching segment timestamp per source (for seek chips)
   957→        if emb_str and seen:
   958→            try:
   959→                seg_rows = (await db.execute(sql_text("""
   960→                    SELECT DISTINCT ON (s.document_id) s.document_id, s.start_time
   961→                    FROM document_segments s
   962→                    WHERE s.document_id = ANY(CAST(:doc_ids AS int[]))
   963→                    ORDER BY s.document_id, s.embedding <=> CAST(:emb AS vector)
   964→                """), {"doc_ids": list(seen), "emb": emb_str})).fetchall()
   965→                seg_map = {r.document_id: round(float(r.start_time or 0), 1) for r in seg_rows}
   966→                for s in sources:
   967→                    if s["document_id"] in seg_map:
   968→                        s["start_time"] = seg_map[s["document_id"]]
   969→            except Exception:
   970→                pass  # no segments table rows; page/doc citations still work
   971→
   972→        # Visual citations: thumbnail chips grouped under their source document
   973→        # (visual-only documents become sources in their own right)
   974→        for v in visual_hits:
   975→            src_doc = next((s for s in sources if s["document_id"] == v["document_id"]), None)
   976→            if src_doc is None:
   977→                src_doc = {"document_id": v["document_id"], "filename": v["filename"],
   978→                           "score": round(float(v.get("score") or 0), 3),
   979→                           "pages": [], "start_time": None}
   980→                sources.append(src_doc)
   981→            src_doc.setdefault("visuals", []).append({
   982→                "asset_id": v["asset_id"],
   983→                "asset_type": v.get("asset_type"),
   984→                "page_number": v.get("page_number"),
   985→                "start_time": v.get("start_time"),
   986→                "labels": v.get("labels") or [],
   987→                "score": round(float(v.get("score") or 0), 3),
   988→            })
   989→            if v.get("page_number") is not None and v["page_number"] not in src_doc["pages"] and len(src_doc["pages"]) < 3:
   990→                src_doc["pages"].append(int(v["page_number"]))
   991→            if v.get("start_time") is not None and src_doc.get("start_time") is None:
   992→                src_doc["start_time"] = round(float(v["start_time"]), 1)
   993→
   994→        # Answer-verified citations (Step 16.5): display only the sources the
   995→        # answer actually references — by [n] marker or by filename — so the
   996→        # sources strip matches what the answer cites. Markers are renumbered
   997→        # to the filtered order. Generic answers (no reference) keep the full
   998→        # evidence-based list, so citations never disappear.
   999→        if sources:
  1000→            referenced: set[int] = set()
  1001→            for num in re.findall(r"\[(\d{1,2})\]", answer):
  1002→                idx = int(num) - 1
  1003→                if 0 <= idx < len(sources):
  1004→                    referenced.add(sources[idx]["document_id"])
  1005→            answer_lower = answer.lower()
  1006→            for s in sources:
  1007→                if s["filename"].lower() in answer_lower:
  1008→                    referenced.add(s["document_id"])
  1009→            if referenced:
  1010→                old_sources = sources
  1011→                sources = [s for s in old_sources if s["document_id"] in referenced]
  1012→                new_index = {s["document_id"]: i + 1 for i, s in enumerate(sources)}
  1013→
  1014→                def _renum(m):
  1015→                    n = int(m.group(1))
  1016→                    if 1 <= n <= len(old_sources):
  1017→                        doc_id = old_sources[n - 1]["document_id"]
  1018→                        if doc_id in new_index:
  1019→                            return f"[{new_index[doc_id]}]"
  1020→                    return ""  # marker points to a dropped source -> remove it
  1021→
  1022→                answer = re.sub(r"\[(\d{1,2})\]", _renum, answer)
  1023→
  1024→        # Step 16.8 (R4 companion): a bare "I don't have enough information"
  1025→        # answer cites nothing — the sources strip must be empty too.
  1026→        if sources and not re.search(r"\[(\d{1,2})\]", answer) and _NO_INFO_RE.match(answer.strip()):
  1027→            sources = []
  1028→
  1029→        pii_map = await get_document_pii_map(db, [s["document_id"] for s in sources])
  1030→        for source in sources:
  1031→            source["pii"] = pii_for_document(pii_map, source["document_id"])
  1032→        pii_notice = build_pii_notice(pii_map, [s["document_id"] for s in sources])
  1033→        if pii_notice["flagged_documents"]:
  1034→            await audit_event(
  1035→                db,
  1036→                None,
  1037→                current_user.id,
  1038→                "CONTENT_PII_FLAGGED_RESULT",
  1039→                f"Chat returned {pii_notice['flagged_documents']} PII-flagged document(s)",
  1040→            )
  1041→        return {"answer": answer, "question": req.question, "sources": sources, "pii_notice": pii_notice, "language": lang}
  1042→    except Exception as e:
  1043→        return {
  1044→            "answer": f"⚠️ AI service unavailable: {str(e)}\n\nYour documents are still searchable. Try using the Search tab instead.",
  1045→            "question": req.question,
  1046→            "error": True
  1047→        }
  1048→

  1135→    """Image embedding -> nearest visual assets. Same SQL contract as

  1162→        for row in label_rows:

  1164→            if len(labels) < 8 and row["label"] not in labels:
  1165→                labels.append(row["label"])

  1168→    return hits
