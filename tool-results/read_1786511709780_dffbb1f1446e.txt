    41→def _vector_dim(cur, table: str, column: str):
    42→    """pgvector dimension of a column, or None if not a vector column."""
    43→    try:
    44→        cur.execute(
    45→            "SELECT atttypmod FROM pg_attribute WHERE attrelid = %s::regclass AND attname = %s",
    46→            (table, column))
    47→        row = cur.fetchone()
    48→        if row and row[0] and row[0] > 0:
    49→            return row[0]
    50→    except Exception:

    99→# ---------------- Chunking (512 tokens / 128 overlap, word-approx) ----------------
   100→
   101→def _hard_split(sentence: str, max_words: int) -> list[str]:
   102→    words = sentence.split()
   103→    return [" ".join(words[i:i + max_words]) for i in range(0, len(words), max_words)]
   104→
   105→def chunk_text(text: str, max_words: int, overlap_words: int) -> list[str]:
   106→    """Sentence-aware chunking. 390 words ~ 512 tokens, 96 words ~ 128 tokens."""
   107→    text = re.sub(r"\n{3,}", "\n\n", (text or "").strip())
   108→    if not text:
   109→        return []
   110→    sentences = re.split(r"(?<=[.!?\u061F\u06D4])\s+|\n+", text)
   111→    sentences = [s.strip() for s in sentences if s and s.strip()]
   112→
   113→    chunks, current, current_words = [], [], 0
   114→    for sent in sentences:
   115→        w = len(sent.split())
   116→        if w > max_words:
   117→            if current:
   118→                chunks.append(" ".join(current))
   119→                current, current_words = [], 0
   120→            chunks.extend(_hard_split(sent, max_words))
   121→            continue
   122→        if current and current_words + w > max_words:
   123→            chunks.append(" ".join(current))
   124→            overlap, kept = [], 0
   125→            for s in reversed(current):
   126→                sw = len(s.split())
   127→                if kept + sw > overlap_words:
   128→                    break
   129→                overlap.insert(0, s)
   130→                kept += sw
   131→            current, current_words = overlap, kept
   132→        current.append(sent)
   133→        current_words += w
   134→    if current:
   135→        chunks.append(" ".join(current))
   136→    return [c for c in chunks if len(c) >= 20] or [text.strip()]

   142→def detect_language(text: str) -> str:
   143→    """'ar' if Arabic characters dominate, else 'en'."""
   144→    sample = (text or "")[:2000]
   145→    arabic = len(re.findall(r"[\u0600-\u06FF]", sample))
   146→    latin = len(re.findall(r"[a-zA-Z]", sample))
   147→    return "ar" if arabic > latin else "en"

   151→def extract_pdf_pages(file_bytes: bytes, on_ocr=None) -> list[dict]:
   152→    """Return [{'page': int, 'text': str}, ...] — text layer, else per-page OCR."""
   153→    import fitz  # PyMuPDF
   154→    doc = fitz.open(stream=file_bytes, filetype="pdf")
   155→    pages = [{"page": i + 1, "text": page.get_text().strip()} for i, page in enumerate(doc)]
   156→    if any(p["text"] for p in pages):
   157→        doc.close()
   158→        return [p for p in pages if p["text"]]
   159→    # Scanned PDF -> render pages and OCR (cap at 30 pages to bound CPU)
   160→    from app.services.ocr_service import extract_text_from_image
   161→    if on_ocr:
   162→        on_ocr()
   163→    out = []
   164→    for i in range(min(len(doc), 30)):
   165→        pix = doc[i].get_pixmap(dpi=150)
   166→        t = extract_text_from_image(pix.tobytes("png"))
   167→        if t.strip():
   168→            out.append({"page": i + 1, "text": t.strip()})
   169→    doc.close()
   170→    return out

   172→def convert_office_to_pdf(file_bytes: bytes, filename: str) -> bytes | None:
   173→    """Step 28: LibreOffice headless Office->PDF conversion (fail-open).
   174→
   175→    Returns the PDF bytes, or None when conversion is impossible (missing
   176→    soffice, corrupt file, timeout). Ingest always continues either way."""
   177→    import subprocess
   178→    import tempfile
   179→    try:
   180→        with tempfile.TemporaryDirectory() as td:
   181→            src = os.path.join(td, os.path.basename(filename))
   182→            with open(src, "wb") as fh:
   183→                fh.write(file_bytes)
   184→            subprocess.run(
   185→                ["soffice", "--headless", "--norestore", "--nolockcheck",
   186→                 "-env:UserInstallation=file:///tmp/lo_profile_seekra",
   187→                 "--convert-to", "pdf", "--outdir", td, src],
   188→                check=True, timeout=180, capture_output=True)
   189→            pdf_path = os.path.splitext(src)[0] + ".pdf"
   190→            with open(pdf_path, "rb") as fh:
   191→                return fh.read()
   192→    except Exception as exc:
   193→        print(f"Office->PDF conversion failed for {filename}: {exc}")
   194→        return None

   197→def extract_docx(file_bytes: bytes) -> str:
   198→    from docx import Document as DocxDocument
   199→    d = DocxDocument(io.BytesIO(file_bytes))
   200→    parts = [p.text for p in d.paragraphs if p.text.strip()]
   201→    for table in d.tables:
   202→        for row in table.rows:
   203→            parts.append(" | ".join(cell.text for cell in row.cells))
   204→    return "\n".join(parts)

   210→def _store_visual_assets(cur, conn, doc_id: int, filename: str, file_bytes: bytes) -> int:
   211→    """Extract, analyze, thumbnail, and store visual assets for one document."""
   212→    import json
   213→    from app.services.minio_client import upload_file
   214→    from app.services.visual_ai import (
   215→        analyze_visual_asset,
   216→        extract_visual_candidates,
   217→        is_visual_candidate,
   218→        release_visual_models,
   219→    )
   220→
   221→    if not is_visual_candidate(filename):
   222→        return 0
   223→
   224→    cur.execute("DELETE FROM visual_assets WHERE document_id = %s", (doc_id,))
   225→    candidates = extract_visual_candidates(filename, file_bytes)
   226→    stored = 0
   227→
   228→    for index, candidate in enumerate(candidates, start=1):
   229→        analyzed = analyze_visual_asset(
   230→            candidate["image"],
   231→            candidate["asset_type"],
   232→            page_number=candidate.get("page_number"),
   233→            start_time=candidate.get("start_time"),
   234→            end_time=candidate.get("end_time"),
   235→            ocr_text=candidate.get("ocr_text"),
   236→        )
   237→        # Free the raw candidate before the next page/frame is analyzed.
   238→        candidate["image"] = None
   239→        thumbnail_path = f"visual/{doc_id}/{index}.jpg"
   240→        upload_file(thumbnail_path, analyzed["thumbnail_bytes"], "image/jpeg")
   241→        embedding = analyzed["embedding"]
   242→        cur.execute(
   243→            """
   244→            INSERT INTO visual_assets
   245→            (document_id, asset_type, page_number, start_time, end_time,
   246→             width, height, ocr_text, thumbnail_path, embedding)
   247→            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, CAST(%s AS vector))
   248→            RETURNING id
   249→            """,
   250→            (
   251→                doc_id,
   252→                analyzed["asset_type"],
   253→                analyzed["page_number"],
   254→                analyzed["start_time"],
   255→                analyzed["end_time"],
   256→                analyzed["width"],
   257→                analyzed["height"],
   258→                analyzed["ocr_text"],
   259→                thumbnail_path,
   260→                str(embedding) if embedding is not None else None,
   261→            ),
   262→        )
   263→        asset_id = cur.fetchone()[0]
   264→        for label in analyzed["labels"]:
   265→            cur.execute(
   266→                """
   267→                INSERT INTO visual_labels (asset_id, label, confidence, bbox)
   268→                VALUES (%s, %s, %s, CAST(%s AS jsonb))
   269→                """,
   270→                (
   271→                    asset_id,
   272→                    label["label"],
   273→                    label["confidence"],
   274→                    json.dumps(label.get("bbox") or []),
   275→                ),
   276→            )
   277→        stored += 1
   278→
   279→    if stored:
   280→        cur.execute("SELECT created_by FROM documents WHERE id = %s", (doc_id,))
   281→        owner = cur.fetchone()
   282→        cur.execute(
   283→            "INSERT INTO audit_logs (user_id, action, details, ip_address) VALUES (%s, %s, %s, %s)",
   284→            (
   285→                owner[0] if owner else None,
   286→                "CONTENT_VISUAL_INDEXED",
   287→                f"Indexed {stored} visual asset(s) for document {doc_id}",
   288→                None,
   289→            ),
   290→        )
   291→    conn.commit()
   292→    release_visual_models()
   293→    return stored

   298→@celery_app.task(bind=True, max_retries=1)
   299→def process_document_task(self, doc_id: int, object_name: str, filename: str):
   300→    from app.services.ocr_service import extract_text_from_image
   301→    from app.services.stt_service import transcribe_audio_segments
   302→    from app.services.embedding import get_embedding
   303→    from app.services.llm_gateway import llm_gateway
   304→    from app.services.minio_client import minio_client
   305→
   306→    conn = _connect()
   307→    cur = conn.cursor()
   308→    segments = []
   309→    pages = None
   310→    content_text = ""
   311→
   312→    try:
   313→        _set_status(cur, conn, doc_id, "processing")
   314→        _set_stage(cur, conn, doc_id, "extracting", 5)
   315→
   316→        response = minio_client.get_object(settings.minio_bucket_name, object_name)
   317→        file_bytes = response.read()
   318→        response.close()
   319→        response.release_conn()
   320→
   321→        lower = filename.lower()
   322→        category = "file"
   323→
   324→        if any(lower.endswith(e) for e in IMAGE_EXTS):
   325→            _set_stage(cur, conn, doc_id, "ocr", 15)
   326→            content_text = extract_text_from_image(file_bytes)
   327→            category = "image"
   328→            _set_stage(cur, conn, doc_id, "ocr", 55)
   329→        elif any(lower.endswith(e) for e in AUDIO_EXTS + VIDEO_EXTS):
   330→            _set_stage(cur, conn, doc_id, "transcribing", 15)
   331→            segments = transcribe_audio_segments(file_bytes, filename)
   332→            content_text = " ".join(s["text"] for s in segments).strip()
   333→            category = "audio" if any(lower.endswith(e) for e in AUDIO_EXTS) else "video"
   334→            _set_stage(cur, conn, doc_id, "transcribing", 60)
   335→        elif lower.endswith(".pdf"):
   336→            _set_stage(cur, conn, doc_id, "extracting", 10)
   337→            pages = extract_pdf_pages(file_bytes, on_ocr=lambda: _set_stage(cur, conn, doc_id, "ocr", 20))
   338→            content_text = "\n".join(p["text"] for p in pages)
   339→            category = "pdf"
   340→            _set_stage(cur, conn, doc_id, "extracting", 55)
   341→        elif lower.endswith(".docx"):
   342→            content_text = extract_docx(file_bytes)
   343→            category = "document"
   344→        elif lower.endswith(".doc"):
   345→            content_text = ""  # legacy binary .doc unsupported
   346→            category = "document"
   347→        elif any(lower.endswith(e) for e in ['.xls', '.xlsx']):
   348→            try:
   349→                import pandas as pd
   350→                content_text = pd.read_excel(io.BytesIO(file_bytes)).to_string()
   351→            except Exception:
   352→                content_text = ""
   353→            category = "spreadsheet"
   354→        elif any(lower.endswith(e) for e in ['.ppt', '.pptx']):
   355→            try:
   356→                from pptx import Presentation
   357→                prs = Presentation(io.BytesIO(file_bytes))
   358→                texts = []
   359→                slide_pages = []
   360→                for si, slide in enumerate(prs.slides, start=1):
   361→                    slide_texts = [shape.text for shape in slide.shapes if hasattr(shape, "text")]
   362→                    texts.extend(slide_texts)
   363→                    st = "\n".join(t for t in slide_texts if t).strip()
   364→                    if st:
   365→                        slide_pages.append({"page": si, "text": st})
   366→                content_text = "\n".join(texts)
   367→                if slide_pages:
   368→                    pages = slide_pages  # chunks get slide numbers as page_number
   369→            except Exception:
   370→                content_text = ""
   371→            category = "presentation"
   372→        else:
   373→            content_text = file_bytes.decode("utf-8", errors="ignore")
   374→            category = "text"
   375→
   376→        # Step 28: build a PDF rendition for Office documents so the viewer
   377→        # can display them with the existing PDF.js path. Stored by naming
   378→        # convention (<object>.preview.pdf) - no schema change required.
   379→        if any(lower.endswith(e) for e in OFFICE_EXTS):
   380→            try:
   381→                _set_stage(cur, conn, doc_id, "converting", 58)
   382→                pdf_bytes = convert_office_to_pdf(file_bytes, filename)
   383→                if pdf_bytes:
   384→                    from app.services.minio_client import upload_file as _upload
   385→                    _upload(object_name + ".preview.pdf", pdf_bytes, "application/pdf")
   386→                    if pages is None:
   387→                        # docx/xlsx had no page numbers; the rendition provides them
   388→                        pages = extract_pdf_pages(pdf_bytes)
   389→                        if pages and not content_text:
   390→                            content_text = "\n".join(p["text"] for p in pages)
   391→            except Exception as exc:
   392→                try:
   393→                    conn.rollback()
   394→                except Exception:
   395→                    pass
   396→                print(f"Document {doc_id}: PDF rendition skipped: {exc}")
   397→
   398→        content_text = (content_text or "").replace("\x00", " ").strip()
   399→
   400→        # Visual assets are indexed before the no-text early return so image-only
   401→        # documents remain searchable even when OCR finds no text.
   402→        visual_count = 0
   403→        try:
   404→            _set_stage(cur, conn, doc_id, "visual", 60)
   405→            visual_count = _store_visual_assets(cur, conn, doc_id, filename, file_bytes)
   406→        except Exception as visual_exc:
   407→            conn.rollback()
   408→            release_visual_models()
   409→            print(f"Document {doc_id}: visual indexing skipped: {visual_exc}")
   410→
   411→        if not content_text and not segments:
   412→            print(f"Document {doc_id}: no text extracted")
   413→            _update_document(cur, conn, doc_id, {
   414→                "content_text": "", "content_type": category,
   415→                "mime_type": category, "summary": "", "status": "completed",
   416→                "stage": "ready", "progress": 100,
   417→                "pii_flagged": False, "pii_count": 0, "pii_risk": "none"})
   418→            _link_default_collection(cur, conn, doc_id)
   419→            return
   420→
   421→        _set_stage(cur, conn, doc_id, "chunking", 65)
   422→
   423→        # ---- Chunk (page-aware) + embed with bge-m3 (1024-dim) ----
   424→        chunk_items = []  # {"text": str, "page": int | None}
   425→        if pages:
   426→            for p in pages:
   427→                for c in chunk_text(_nfkc(p["text"].replace("\x00", " ")), settings.chunk_max_words, settings.chunk_overlap_words):
   428→                    chunk_items.append({"text": c, "page": p["page"]})
   429→        else:
   430→            for c in chunk_text(content_text, settings.chunk_max_words, settings.chunk_overlap_words):
   431→                chunk_items.append({"text": c, "page": None})
   432→        pii_chunk_findings = []
   433→        for idx, item in enumerate(chunk_items):
   434→            for finding in detect_pii(item["text"]):
   435→                finding["chunk_index"] = idx
   436→                finding["page_number"] = item.get("page")
   437→                pii_chunk_findings.append(finding)
   438→
   439→        print(f"Document {doc_id}: {len(chunk_items)} chunks, {len(segments)} A/V segments")
   440→
   441→        cur.execute("DELETE FROM document_chunks WHERE document_id = %s", (doc_id,))
   442→        cur.execute("DELETE FROM document_segments WHERE document_id = %s", (doc_id,))
   443→        cur.execute("DELETE FROM pii_findings WHERE document_id = %s", (doc_id,))
   444→
   445→        pii_total = len(pii_chunk_findings)
   446→        pii_risk = highest_risk(finding["risk_level"] for finding in pii_chunk_findings)
   447→        for finding in pii_chunk_findings:
   448→            cur.execute(
   449→                "INSERT INTO pii_findings (document_id, chunk_index, page_number, pii_type, risk_level, match_count, masked_sample) "
   450→                "VALUES (%s, %s, %s, %s, %s, %s, %s)",
   451→                (doc_id, finding["chunk_index"], finding["page_number"], finding["pii_type"],
   452→                 finding["risk_level"], 1, finding["masked_sample"]))
   453→        if pii_total:
   454→            cur.execute("SELECT created_by FROM documents WHERE id = %s", (doc_id,))
   455→            owner = cur.fetchone()
   456→            cur.execute(
   457→                "INSERT INTO audit_logs (user_id, action, details, ip_address) VALUES (%s, %s, %s, %s)",
   458→                (owner[0] if owner else None, "CONTENT_PII_DETECTED",
   459→                 f"Detected {pii_total} PII finding(s) in document {doc_id}; max risk: {pii_risk}", None))
   460→        conn.commit()
   461→
   462→        chunk_embeddings = []
   463→        for idx, item in enumerate(chunk_items):
   464→            emb = get_embedding(item["text"])
   465→            chunk_embeddings.append(emb)
   466→            cur.execute(
   467→                "INSERT INTO document_chunks (document_id, chunk_text, chunk_index, page_number, embedding) "
   468→                "VALUES (%s, %s, %s, %s, %s::vector)",
   469→                (doc_id, item["text"], idx, item["page"], str(emb)))
   470→            if (idx + 1) % 5 == 0 or idx + 1 == len(chunk_items):
   471→                conn.commit()
   472→                _set_stage(cur, conn, doc_id, "embedding", 65 + int(20 * (idx + 1) / max(1, len(chunk_items))))
   473→                print(f"Document {doc_id}: embedded chunk {idx + 1}/{len(chunk_items)}")
   474→
   475→        # ---- Timestamped segments (audio/video) ----
   476→        for i, seg in enumerate(segments):
   477→            seg["text"] = (seg["text"] or "").replace("\x00", " ")
   478→            emb = get_embedding(seg["text"])
   479→            cur.execute(
   480→                "INSERT INTO document_segments (document_id, segment_text, start_time, end_time, embedding) "
   481→                "VALUES (%s, %s, %s, %s, %s::vector)",
   482→                (doc_id, seg["text"], seg["start"], seg["end"], str(emb)))
   483→            if (i + 1) % 20 == 0 or i + 1 == len(segments):
   484→                conn.commit()
   485→                _set_stage(cur, conn, doc_id, "embedding", 85 + int(10 * (i + 1) / max(1, len(segments))))
   486→                print(f"Document {doc_id}: embedded segment {i + 1}/{len(segments)}")
   487→
   488→        _set_stage(cur, conn, doc_id, "summarizing", 95)
   489→
   490→        # ---- Summary via Qwen (graceful fallback) ----
   491→        try:
   492→            doc_lang = detect_language(content_text)
   493→            # Full-coverage summary input: opening + closing, not just the first
   494→            # 3000 chars (same quality bar as on-demand summaries, Step 17A.2)
   495→            summary_input = content_text if len(content_text) <= 4000 else (
   496→                content_text[:2500] + "\n[...]\n" + content_text[-1500:]
   497→            )
   498→            summary = asyncio.run(llm_gateway.summarize(summary_input, doc_lang))
   499→        except Exception as e:
   500→            print(f"AI summary failed (fallback to excerpt): {e}")
   501→            summary = content_text[:500] + ("..." if len(content_text) > 500 else "")
   502→
   503→        # ---- Update document row ----
   504→        _update_document(cur, conn, doc_id, {
   505→            "content_text": content_text, "content_type": category,
   506→            "mime_type": category, "summary": summary, "language": doc_lang, "status": "completed",
   507→            "stage": "ready", "progress": 100,
   508→            "pii_flagged": pii_total > 0, "pii_count": pii_total, "pii_risk": pii_risk})
   509→
   510→        # Optional: doc-level mean embedding, only if legacy vector column is 1024-dim
   511→        try:
   512→            if "embedding" in _table_columns(cur, "documents") and chunk_embeddings:
   513→                dim = _vector_dim(cur, "documents", "embedding")
   514→                if dim == len(chunk_embeddings[0]):
   515→                    mean_emb = [sum(v) / len(v) for v in zip(*chunk_embeddings)]
   516→                    norm = math.sqrt(sum(x * x for x in mean_emb)) or 1.0
   517→                    mean_emb = [x / norm for x in mean_emb]
   518→                    cur.execute("UPDATE documents SET embedding = %s::vector WHERE id = %s",
   519→                                (str(mean_emb), doc_id))
   520→                    conn.commit()
   521→        except Exception as e:
   522→            conn.rollback()
   523→            print(f"Doc-level embedding skipped: {e}")
   524→
   525→        _link_default_collection(cur, conn, doc_id)
   526→        print(f"Document {doc_id} DONE: {len(chunk_items)} chunks, {len(segments)} segments, {len(content_text)} chars")
   527→
   528→    except Exception as exc:
   529→        conn.rollback()
   530→        print(f"Document {doc_id} FAILED: {exc}")
   531→        _set_status(cur, conn, doc_id, "failed")
   532→        _set_stage(cur, conn, doc_id, "failed")
   533→        raise self.retry(exc=exc, countdown=60)
   534→    finally:
   535→        cur.close()
   536→        conn.close()

   539→# ---------------- Step 12B: existing-document PII rescan ----------------

   541→@celery_app.task(bind=True, max_retries=1, name="app.workers.celery_tasks.rescan_pii_task")
   542→def rescan_pii_task(self):
   543→    """Rescan indexed chunks and rebuild masked PII findings for all documents."""
   544→    conn = _connect()
   545→    cur = conn.cursor()
   546→    scanned = 0
   547→    flagged_documents = 0
   548→    total_findings = 0
   549→    try:
   550→        cur.execute("SELECT id, created_by FROM documents ORDER BY id")
   551→        documents = cur.fetchall()
   552→        for doc_id, owner_id in documents:
   553→            cur.execute(
   554→                "SELECT chunk_index, page_number, chunk_text FROM document_chunks WHERE document_id = %s ORDER BY chunk_index",
   555→                (doc_id,))
   556→            rows = cur.fetchall()
   557→            findings = []
   558→            for chunk_index, page_number, chunk_text in rows:
   559→                for finding in detect_pii(chunk_text):
   560→                    finding["chunk_index"] = chunk_index
   561→                    finding["page_number"] = page_number
   562→                    findings.append(finding)
   563→
   564→            cur.execute("DELETE FROM pii_findings WHERE document_id = %s", (doc_id,))
   565→            doc_findings = len(findings)
   566→            doc_risk = highest_risk(finding["risk_level"] for finding in findings)
   567→            for finding in findings:
   568→                cur.execute(
   569→                    "INSERT INTO pii_findings (document_id, chunk_index, page_number, pii_type, risk_level, match_count, masked_sample) "
   570→                    "VALUES (%s, %s, %s, %s, %s, %s, %s)",
   571→                    (doc_id, finding["chunk_index"], finding["page_number"], finding["pii_type"],
   572→                     finding["risk_level"], 1, finding["masked_sample"]))
   573→            cur.execute(
   574→                "UPDATE documents SET pii_flagged = %s, pii_count = %s, pii_risk = %s, updated_at = NOW() WHERE id = %s",
   575→                (doc_findings > 0, doc_findings, doc_risk, doc_id))
   576→            if doc_findings:
   577→                cur.execute(
   578→                    "INSERT INTO audit_logs (user_id, action, details, ip_address) VALUES (%s, %s, %s, %s)",
   579→                    (owner_id, "CONTENT_PII_DETECTED",
   580→                     f"Rescan detected {doc_findings} PII finding(s) in document {doc_id}; max risk: {doc_risk}", None))
   581→            conn.commit()
   582→            scanned += 1
   583→            if doc_findings:
   584→                flagged_documents += 1
   585→                total_findings += doc_findings
   586→        return {
   587→            "scanned_documents": scanned,
   588→            "flagged_documents": flagged_documents,
   589→            "total_findings": total_findings,
   590→        }
   591→    except Exception as exc:
   592→        conn.rollback()
   593→        print(f"PII rescan FAILED: {exc}")
   594→        raise self.retry(exc=exc, countdown=60)
   595→    finally:
   596→        cur.close()
   597→        conn.close()
