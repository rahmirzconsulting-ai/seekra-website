    48→function formatTime(seconds: number): string {
    49→  const total = Math.floor(seconds);

    54→function HighlightedText({ text, query }: { text: string; query: string }) {
    55→  if (!text || !query) return <span>{text}</span>;
    56→  const terms = query.trim().split(/\s+/).filter((term) => term.length > 1);
    57→  if (!terms.length) return <span>{text}</span>;
    58→  let parts: (string | JSX.Element)[] = [text];
    59→  terms.forEach((term, termIndex) => {
    60→    const next: (string | JSX.Element)[] = [];
    61→    const escaped = term.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    62→    const regex = new RegExp(`(${escaped})`, "gi");
    63→    parts.forEach((part) => {
    64→      if (typeof part !== "string") {
    65→        next.push(part);
    66→        return;
    67→      }
    68→      part.split(regex).forEach((piece, pieceIndex) => {
    69→        if (pieceIndex % 2 === 1) {
    70→          next.push(
    71→            <mark key={`${termIndex}-${pieceIndex}-${piece}`} className="bg-primary/25 text-foreground rounded px-0.5">
    72→              {piece}
    73→            </mark>
    74→          );
    75→        } else if (piece) {
    76→          next.push(piece);
    77→        }
    78→      });
    79→    });
    80→    parts = next;
    81→  });
    82→  return <>{parts}</>;
    83→}

   120→function resultSource(result: SmartSearchResult): Exclude<SourceFilter, "all"> {
   121→  if (result.result_type === "visual_asset") {
   122→    if (result.asset_type === "pdf_page") return "pdf_pages";
   123→    if (result.asset_type === "video_frame") return "videos";
   124→    return "images";
   125→  }
   126→  // 22B: classify by the real file type, not "everything is a document" -
   127→  // an MP3 transcript hit is audio, a video transcript hit is a video.
   128→  const ct = (result.content_type || "").toLowerCase();
   129→  if (ct === "audio" || ct.startsWith("audio/")) return "audio";
   130→  if (ct === "video" || ct.startsWith("video/")) return "videos";
   131→  if (ct === "image" || ct.startsWith("image/")) return "images";
   132→  return "documents";
   133→}

   135→function sourceLabel(result: SmartSearchResult, tf: TranslateWithFallback): string {
   136→  if (result.result_type === "visual_asset") {
   137→    if (result.asset_type === "pdf_page") {
   138→      return `${tf("search.sourcePdfPage", "PDF page")} ${result.page_number || ""}`.trim();
   139→    }
   140→    if (result.asset_type === "video_frame") {
   141→      return `${tf("search.sourceVideo", "Video")} ${formatTime(result.start_time || 0)}`;
   142→    }
   143→    return tf("search.sourceImage", "Image");
   144→  }
   145→  const contentType = (result.content_type || "").toLowerCase();
   146→  if (contentType.includes("video")) return tf("search.sourceVideo", "Video");
   147→  if (contentType.includes("audio")) return tf("search.sourceAudio", "Audio");
   148→  if (contentType.includes("pdf")) return tf("search.sourcePdf", "PDF");
   149→  if (contentType.startsWith("image/")) return tf("search.sourceImage", "Image");
   150→  return tf("search.sourceDocument", "Document");
   151→}

   153→function localizedMatchReason(reason: string, tf: TranslateWithFallback): string {
   154→  const reasonKeys: Record<string, [string, string]> = {
   155→    "Filename match": ["search.reasonFilename", "Filename match"],
   156→    "Text match": ["search.reasonText", "Text match"],
   157→    "Transcript match": ["search.reasonTranscript", "Transcript match"],
   158→    "Semantic match": ["search.reasonSemantic", "Semantic match"],
   159→    "Semantic audio match": ["search.reasonSemanticAudio", "Semantic audio match"],
   160→    "Text inside image": ["search.reasonVisualText", "Text inside image"],
   161→    "Object match": ["search.reasonObject", "Object match"],
   162→    "Visual match": ["search.reasonVisual", "Visual match"],
   163→  };
   164→  const [key, fallback] = reasonKeys[reason] || ["", reason];
   165→  return key ? tf(key, fallback) : reason;
   166→}
