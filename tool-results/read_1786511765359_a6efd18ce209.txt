    99→    99→# ---------------- Chunking (512 tokens / 128 overlap, word-approx) ----------------
   100→   100→
   101→   101→def _hard_split(sentence: str, max_words: int) -> list[str]:
   102→   102→    words = sentence.split()
   103→   103→    return [" ".join(words[i:i + max_words]) for i in range(0, len(words), max_words)]

   105→   105→def chunk_text(text: str, max_words: int, overlap_words: int) -> list[str]:
   106→   106→    """Sentence-aware chunking. 390 words ~ 512 tokens, 96 words ~ 128 tokens."""
   107→   107→    text = re.sub(r"\n{3,}", "\n\n", (text or "").strip())
   108→   108→    if not text:
   109→   109→        return []
   110→   110→    sentences = re.split(r"(?<=[.!?\u061F\u06D4])\s+|\n+", text)
   111→   111→    sentences = [s.strip() for s in sentences if s and s.strip()]
   112→   112→
   113→   113→    chunks, current, current_words = [], [], 0
   114→   114→    for sent in sentences:
   115→   115→        w = len(sent.split())
   116→   116→        if w > max_words:
   117→   117→            if current:
   118→   118→                chunks.append(" ".join(current))
   119→   119→                current, current_words = [], 0
   120→   120→            chunks.extend(_hard_split(sent, max_words))
   121→   121→            continue
   122→   122→        if current and current_words + w > max_words:
   123→   123→            chunks.append(" ".join(current))
   124→   124→            overlap, kept = [], 0
   125→   125→            for s in reversed(current):
   126→   126→                sw = len(s.split())
   127→   127→                if kept + sw > overlap_words:
   128→   128→                    break
   129→   129→                overlap.insert(0, s)
   130→   130→                kept += sw
   131→   131→            current, current_words = overlap, kept
   132→   132→        current.append(sent)
   133→   133→        current_words += w
   134→   134→    if current:
   135→   135→        chunks.append(" ".join(current))
   136→   136→    return [c for c in chunks if len(c) >= 20] or [text.strip()]

   142→   142→def detect_language(text: str) -> str:
   143→   143→    """'ar' if Arabic characters dominate, else 'en'."""
   144→   144→    sample = (text or "")[:2000]
   145→   145→    arabic = len(re.findall(r"[\u0600-\u06FF]", sample))
   146→   146→    latin = len(re.findall(r"[a-zA-Z]", sample))
   147→   147→    return "ar" if arabic > latin else "en"

   151→   151→def extract_pdf_pages(file_bytes: bytes, on_ocr=None) -> list[dict]:
   152→   152→    """Return [{'page': int, 'text': str}, ...] — text layer, else per-page OCR."""

   172→   172→def convert_office_to_pdf(file_bytes: bytes, filename: str) -> bytes | None:
   173→   173→    """Step 28: LibreOffice headless Office->PDF conversion (fail-open).
   174→   174→
   175→   175→    Returns the PDF bytes, or None when conversion is impossible (missing
   176→   176→    soffice, corrupt file, timeout). Ingest always continues either way."""
   177→   177→    import subprocess
   178→   178→    import tempfile
   179→   179→    try:
   180→   180→        with tempfile.TemporaryDirectory() as td:
   181→   181→            src = os.path.join(td, os.path.basename(filename))
   182→   182→            with open(src, "wb") as fh:
   183→   183→                fh.write(file_bytes)
   184→   184→            subprocess.run(
   185→   185→                ["soffice", "--headless", "--norestore", "--nolockcheck",
   186→   186→                 "-env:UserInstallation=file:///tmp/lo_profile_seekra",
   187→   187→                 "--convert-to", "pdf", "--outdir", td, src],
   188→   188→                check=True, timeout=180, capture_output=True)
   189→   189→            pdf_path = os.path.splitext(src)[0] + ".pdf"
   190→   190→            with open(pdf_path, "rb") as fh:
   191→   191→                return fh.read()
   192→   192→    except Exception as exc:
   193→   193→        print(f"Office->PDF conversion failed for {filename}: {exc}")
   194→   194→        return None

   439→   439→        print(f"Document {doc_id}: {len(chunk_items)} chunks, {len(segments)} A/V segments")
   440→   440→
   441→   441→        cur.execute("DELETE FROM document_chunks WHERE document_id = %s", (doc_id,))
   442→   442→        cur.execute("DELETE FROM document_segments WHERE document_id = %s", (doc_id,))
   443→   443→        cur.execute("DELETE FROM pii_findings WHERE document_id = %s", (doc_id,))
   444→   444→
   445→   445→        pii_total = len(pii_chunk_findings)
   446→   446→        pii_risk = highest_risk(finding["risk_level"] for finding in pii_chunk_findings)
   447→   447→        for finding in pii_chunk_findings:
   448→   448→            cur.execute(
   449→   449→                "INSERT INTO pii_findings (document_id, chunk_index, page_number, pii_type, risk_level, match_count, masked_sample) "

   454→   454→            cur.execute("SELECT created_by FROM documents WHERE id = %s", (doc_id,))
   455→   455→            owner = cur.fetchone()
   456→   456→            cur.execute(
   457→   457→                "INSERT INTO audit_logs (user_id, action, details, ip_address) VALUES (%s, %s, %s, %s)",
   458→   458→                (owner[0] if owner else None, "CONTENT_PII_DETECTED",
   459→   459→                 f"Detected {pii_total} PII finding(s) in document {doc_id}; max risk: {pii_risk}", None))
   460→   460→        conn.commit()
   461→   461→
   462→   462→        chunk_embeddings = []
   463→   463→        for idx, item in enumerate(chunk_items):
   464→   464→            emb = get_embedding(item["text"])
   465→   465→            chunk_embeddings.append(emb)
   466→   466→            cur.execute(
   467→   467→                "INSERT INTO document_chunks (document_id, chunk_text, chunk_index, page_number, embedding) "
   468→   468→                "VALUES (%s, %s, %s, %s, %s::vector)",
   469→   469→                (doc_id, item["text"], idx, item["page"], str(emb)))
   470→   470→            if (idx + 1) % 5 == 0 or idx + 1 == len(chunk_items):
   471→   471→                conn.commit()
   472→   472→                _set_stage(cur, conn, doc_id, "embedding", 65 + int(20 * (idx + 1) / max(1, len(chunk_items))))
   473→   473→                print(f"Document {doc_id}: embedded chunk {idx + 1}/{len(chunk_items)}")
   474→   474→
   475→   475→        # ---- Timestamped segments (audio/video) ----
   476→   476→        for i, seg in enumerate(segments):
   477→   477→            seg["text"] = (seg["text"] or "").replace("\x00", " ")
   478→   478→            emb = get_embedding(seg["text"])
   479→   479→            cur.execute(
   480→   480→                "INSERT INTO document_segments (document_id, segment_text, start_time, end_time, embedding) "
   481→   481→                "VALUES (%s, %s, %s, %s, %s::vector)",
   482→   482→                (doc_id, seg["text"], seg["start"], seg["end"], str(emb)))
   483→   483→            if (i + 1) % 20 == 0 or i + 1 == len(segments):
   484→   484→                conn.commit()
   485→   485→                _set_stage(cur, conn, doc_id, "embedding", 85 + int(10 * (i + 1) / max(1, len(segments))))
   486→   486→                print(f"Document {doc_id}: embedded segment {i + 1}/{len(segments)}")
   487→   487→
   488→   488→        _set_stage(cur, conn, doc_id, "summarizing", 95)
   489→   489→
   490→   490→        # ---- Summary via Qwen (graceful fallback) ----
   491→   491→        try:
   492→   492→            doc_lang = detect_language(content_text)
   493→   493→            # Full-coverage summary input: opening + closing, not just the first
   494→   494→            # 3000 chars (same quality bar as on-demand summaries, Step 17A.2)

   499→   499→        except Exception as e:

   501→   501→            summary = content_text[:500] + ("..." if len(content_text) > 500 else "")
   502→   502→
   503→   503→        # ---- Update document row ----
   504→   504→        _update_document(cur, conn, doc_id, {
   505→   505→            "content_text": content_text, "content_type": category,
   506→   506→            "mime_type": category, "summary": summary, "language": doc_lang, "status": "completed",
   507→   507→            "stage": "ready", "progress": 100,
   508→   508→            "pii_flagged": pii_total > 0, "pii_count": pii_total, "pii_risk": pii_risk})
   509→   509→
   510→   510→        # Optional: doc-level mean embedding, only if legacy vector column is 1024-dim
   511→   511→        try:
   512→   512→            if "embedding" in _table_columns(cur, "documents") and chunk_embeddings:
   513→   513→                dim = _vector_dim(cur, "documents", "embedding")
   514→   514→                if dim == len(chunk_embeddings[0]):
   515→   515→                    mean_emb = [sum(v) / len(v) for v in zip(*chunk_embeddings)]
   516→   516→                    norm = math.sqrt(sum(x * x for x in mean_emb)) or 1.0
   517→   517→                    mean_emb = [x / norm for x in mean_emb]
   518→   518→                    cur.execute("UPDATE documents SET embedding = %s::vector WHERE id = %s",
   519→   519→                                (str(mean_emb), doc_id))
   520→   520→                    conn.commit()
   521→   521→        except Exception as e:
   522→   522→            conn.rollback()
   523→   523→            print(f"Doc-level embedding skipped: {e}")
   524→   524→
   525→   525→        _link_default_collection(cur, conn, doc_id)
   526→   526→        print(f"Document {doc_id} DONE: {len(chunk_items)} chunks, {len(segments)} segments, {len(content_text)} chars")
   527→   527→
   528→   528→    except Exception as exc:
   529→   529→        conn.rollback()
   530→   530→        print(f"Document {doc_id} FAILED: {exc}")
   531→   531→        _set_status(cur, conn, doc_id, "failed")
   532→   532→        _set_stage(cur, conn, doc_id, "failed")
   533→   533→        raise self.retry(exc=exc, countdown=60)
   534→   534→    finally:
   535→   535→        cur.close()
   536→   536→        conn.close()
