   131→   131→def detect_language(text: str) -> str:
   132→   132→    sample = (text or "")[:1000]
   133→   133→    # Step 30E: Urdu-specific Nastaliq letters (never used in Arabic):
   134→   134→    # ٹ ڈ ڑ ں ھ ہ ے
   135→   135→    urdu = len(re.findall(r"[\u0679\u0688\u0691\u06BA\u06BE\u06C1\u06D2]", sample))
   136→   136→    if urdu > 0:
   137→   137→        return "ur"
   138→   138→    arabic = len(re.findall(r"[\u0600-\u06FF]", sample))
   139→   139→    latin = len(re.findall(r"[a-zA-Z]", sample))
   140→   140→    return "ar" if arabic > latin else "en"

   174→   174→async def _contextualize_question(question: str, history: list[dict], lang: str) -> str:
   175→   175→    """Rewrite a follow-up into a standalone retrieval query (fail-open)."""
   176→   176→    if not history or not _looks_like_followup(question):
   177→   177→        return question
   178→   178→    lang_name = LANG_NAMES.get(lang, "English")
   179→   179→    prompt = (
   180→   180→        "You rewrite follow-up questions into standalone search questions for a "
   181→   181→        "document search system. Given the conversation and the follow-up, output "
   182→   182→        "ONLY the rewritten standalone question in " + lang_name + ". Resolve "
   183→   183→        "references like 'this book', 'it', 'that document' to the specific "
   184→   184→        "document or topic named earlier in the conversation. Do NOT answer the "
   185→   185→        "question. If the follow-up is already standalone, output it unchanged.\n\n"
   186→   186→        "Conversation:\n" + _format_history(history) + "\n\n"
   187→   187→        "Follow-up: " + question + "\n\nStandalone question:"
   188→   188→    )
   189→   189→    try:
   190→   190→        rewritten = await asyncio.wait_for(
   191→   191→            llm_gateway._generate(prompt, max_tokens=60, kind="chat"),
   192→   192→            timeout=REWRITE_TIMEOUT)
   193→   193→        rewritten = unicodedata.normalize("NFKC", (rewritten or "").strip().strip('"').strip())
   194→   194→        first_line = rewritten.splitlines()[0].strip() if rewritten else ""
   195→   195→        if first_line and len(first_line) <= 400:
   196→   196→            return first_line
   197→   197→    except Exception:
   198→   198→        pass  # timeout / provider error -> original question
   199→   199→    return question

   223→   223→            or ext in ("doc", "docx", "ppt", "pptx", "xls", "xlsx", "csv"):
   224→   224→        return "Office"
   225→   225→    if ct.startswith("text") or ext in ("txt", "md"):

   230→   230→async def resolve_target_documents(question: str, db: AsyncSession, user=None) -> list[int]:
   231→   231→    """Resolve meta-references: 'latest document' intent + filename mentions."""
   232→   232→    q_lower = question.lower()
   233→   233→    perm_params: dict = {}
   234→   234→    perm_sql = permission_clause(user, perm_params, "documents")
   235→   235→
   236→   236→    # 1. "latest / newest document" intent (EN + AR)
   237→   237→    if any(m in q_lower for m in LATEST_MARKERS):
   238→   238→        result = await db.execute(sql_text(
   239→   239→            f"SELECT id FROM documents WHERE {perm_sql} ORDER BY created_at DESC LIMIT 1"), perm_params)
   240→   240→        row = result.first()
   241→   241→        return [row[0]] if row else []
   242→   242→
   243→   243→    # 2. Filename mention detection (distinctive tokens, e.g. "honda", "mant aq")
   244→   244→    result = await db.execute(sql_text(f"SELECT id, filename FROM documents WHERE {perm_sql}"), perm_params)
   245→   245→    q_tokens = set(re.findall(r"\w+", q_lower)) - STOPWORDS
   246→   246→    matched = []
   247→   247→    for row in result.fetchall():
   248→   248→        name_tokens = set(re.sub(r"[._\-]", " ", row.filename.lower()).split()) - STOPWORDS
   249→   249→        overlap = name_tokens & q_tokens
   250→   250→        distinctive = {t for t in overlap if len(t) >= 4 and t not in GENERIC_NAME_WORDS}
   251→   251→        # scope only on strong evidence: >=2 shared tokens, or one distinctive token
   252→   252→        if len(overlap) >= 2 or (len(overlap) == 1 and distinctive):
   253→   253→            matched.append(row.id)
   254→   254→    return matched

   321→   321→              AND to_tsvector('simple', coalesce(va.ocr_text, '')) @@ websearch_to_tsquery('simple', :query){scope}
   322→   322→            ORDER BY score DESC, va.id

   350→   350→async def _visual_vector_leg(question: str, target_ids: list[int], fetch: int = 8, user=None) -> list[dict]:
   351→   351→    embedding = await _cached_visual_embedding(question)
   352→   352→    if not embedding:
   353→   353→        return []
   354→   354→    params: dict = {"emb": str(embedding), "fetch": fetch}
   355→   355→    scope = _visual_scope(target_ids, params)
   356→   356→    scope += " AND (" + permission_clause(user, params, "d") + ")"
   357→   357→    async with async_session() as session:
   358→   358→        rows = (await session.execute(sql_text(f"""
   359→   359→            SELECT va.id AS asset_id, va.document_id, va.asset_type, va.page_number,
   360→   360→                   va.start_time, va.end_time, va.ocr_text,
   361→   361→                   1 - (va.embedding <=> CAST(:emb AS vector)) AS score,
   362→   362→                   d.filename
   363→   363→            FROM visual_assets va
   364→   364→            JOIN documents d ON d.id = va.document_id
   365→   365→            WHERE COALESCE(d.status, 'completed') = 'completed'
   366→   366→              AND va.embedding IS NOT NULL{scope}
   367→   367→            ORDER BY va.embedding <=> CAST(:emb AS vector)
