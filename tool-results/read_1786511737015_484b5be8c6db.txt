    31→// 18B6: the file-type icon doubles as a real thumbnail for images and a
    32→// click-through to the viewer for every type.
    33→function FileThumb({
    34→  file,
    35→  Icon,
    36→  onOpen,
    37→}: {
    38→  file: DocumentResponse;
    39→  Icon: ComponentType<{ className?: string }>;
    40→  onOpen: () => void;
    41→}) {
    42→  const [src, setSrc] = useState<string | null>(null);
    43→
    44→  useEffect(() => {
    45→    if (!file.mime_type || !file.mime_type.startsWith("image/")) return;
    46→    let url: string | null = null;
    47→    let cancelled = false;
    48→    downloadFile(file.id, file.filename, false)
    49→      .then((blob) => {
    50→        if (cancelled) return;
    51→        url = URL.createObjectURL(blob);
    52→        setSrc(url);
    53→      })
    54→      .catch(() => undefined);
    55→    return () => {
    56→      cancelled = true;
    57→      if (url) URL.revokeObjectURL(url);
    58→    };
    59→  }, [file.id, file.filename, file.mime_type]);
    60→
    61→  return (
    62→    <button
    63→      onClick={onOpen}
    64→      title={file.filename}
    65→      className="shrink-0 rounded-lg overflow-hidden hover:ring-2 hover:ring-primary/50 transition-all cursor-pointer"
    66→    >
    67→      {src ? (
    68→        // eslint-disable-next-line @next/next/no-img-element
    69→        <img src={src} alt={file.filename} className="w-8 h-8 object-cover" />
    70→      ) : (
    71→        <div className="p-1.5 rounded-lg bg-surface">
    72→          <Icon className="w-4 h-4 text-primary" />
    73→        </div>
    74→      )}
    75→    </button>
    76→  );
    77→}

   560→            <div className="absolute inset-0 bg-black/50 backdrop-blur-sm" onClick={() => !bulkBusy && setBulkDeleteOpen(false)} />

   564→                {selected.size} {t("files.bulkDeleteMsg")}
   565→              </p>
   566→              <p className="mt-1 text-sm text-destructive">{t("files.deleteUndo")}</p>

   568→                <p className="mt-3 p-2 rounded-lg bg-destructive/10 border border-destructive/20 text-destructive text-sm">{bulkError}</p>

   570→              <div className="mt-6 flex items-center justify-end gap-3">

   576→                  {t("files.cancel")}
   577→                </button>

   579→                  onClick={handleBulkDelete}
   580→                  disabled={bulkBusy}
   581→                  className="px-4 py-2 rounded-lg bg-destructive text-destructive-foreground text-sm font-medium hover:opacity-90 transition-all disabled:opacity-50 flex items-center gap-2"

   590→
   591→        {/* Move-to-collection modal */}

   602→                  {tf("files.moveNoSelection", "No files selected yet — tick the checkboxes next to the files you want to move, then pick the destination collection here.")}
   603→                </p>
   604→              )}
   605→              <div className="mt-4 space-y-1 max-h-64 overflow-y-auto">
   606→                <button
   607→                  onClick={() => setMoveTarget(null)}

   610→                      ? "bg-primary/10 border-primary/40 text-primary font-medium"

   614→                  {t("files.bulkMoveRoot")}

   632→              </div>

   636→              <div className="mt-6 flex items-center justify-end gap-3">
   637→                <button
   638→                  onClick={() => setMoveOpen(false)}
   639→                  disabled={bulkBusy}
   640→                  className="px-4 py-2 rounded-lg bg-surface border border-border text-sm font-medium text-foreground hover:bg-muted transition-all disabled:opacity-50"
   641→                >
   642→                  {t("files.cancel")}
   643→                </button>
   644→                <button
   645→                  onClick={handleBulkMove}
   646→                  disabled={bulkBusy || selected.size === 0}
   647→                  className="px-4 py-2 rounded-lg bg-primary text-primary-foreground text-sm font-medium hover:bg-primary/90 transition-all disabled:opacity-50 flex items-center gap-2"
   648→                >
   649→                  {bulkBusy && <Loader2 className="w-4 h-4 animate-spin" />}
   650→                  {t("files.bulkMove")}
   651→                </button>
   652→              </div>
   653→            </div>
   654→          </div>
